package com.example.demo.annotation;

import java.lang.annotation.*;

@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface MaskResponse {
    String[] fields();
    String maskValue() default "***";
}

    }
}

// src/main/java/com/example/demo/handler/AnnotationDrivenMaskingHandler.java
package com.example.demo.handler;

import com.example.demo.annotation.MaskResponse;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.*;
import org.springframework.http.MediaType;
import org.springframework.http.codec.HttpMessageWriter;
import org.springframework.web.reactive.result.method.HandlerResult;
import org.springframework.web.reactive.result.method.HandlerResultHandler;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

import java.lang.reflect.Method;
import java.util.*;

public class AnnotationDrivenMaskingHandler implements HandlerResultHandler {

    private final List<HttpMessageWriter<?>> writers;
    private final ObjectMapper objectMapper;

    public AnnotationDrivenMaskingHandler(List<HttpMessageWriter<?>> writers, ObjectMapper objectMapper) {
        this.writers = writers;
        this.objectMapper = objectMapper;
    }

    @Override
    public boolean supports(HandlerResult result) {
        return true;
    }

    @Override
    public Mono<Void> handleResult(ServerWebExchange exchange, HandlerResult result) {
        Method method = result.getHandlerMethod().getMethod();
        MaskResponse annotation = method.getAnnotation(MaskResponse.class);

        if (annotation == null) {
            return defaultWrite(exchange, result);
        }

        String[] maskedFields = annotation.fields();
        String maskValue = annotation.maskValue();

        Object returnValue = result.getReturnValue();
        if (returnValue instanceof Mono) {
            return ((Mono<?>) returnValue)
                    .map(val -> objectMapper.valueToTree(val))
                    .map(node -> maskAndClean(node, maskedFields, maskValue))
                    .flatMap(modified -> write(exchange, modified, result));
        }

        JsonNode node = objectMapper.valueToTree(returnValue);
        JsonNode cleaned = maskAndClean(node, maskedFields, maskValue);
        return write(exchange, cleaned, result);
    }

    private Mono<Void> defaultWrite(ServerWebExchange exchange, HandlerResult result) {
        Object returnValue = result.getReturnValue();
        if (returnValue instanceof Mono) {
            return ((Mono<?>) returnValue)
                    .map(val -> objectMapper.valueToTree(val))
                    .map(this::removeNullsRecursively)
                    .flatMap(modified -> write(exchange, modified, result));
        }
        return write(exchange, objectMapper.valueToTree(returnValue), result);
    }

    private Mono<Void> write(ServerWebExchange exchange, JsonNode node, HandlerResult result) {
        @SuppressWarnings("unchecked")
        HttpMessageWriter<Object> writer = (HttpMessageWriter<Object>) writers.stream()
                .filter(w -> w.canWrite(result.getReturnTypeSource(), MediaType.APPLICATION_JSON))
                .findFirst()
                .orElseThrow(() -> new IllegalStateException("No writer found"));

        return writer.write(Mono.just(node), node.getClass(), MediaType.APPLICATION_JSON, exchange.getResponse(), null);
    }

    private JsonNode maskAndClean(JsonNode node, String[] paths, String maskValue) {
        ObjectNode masked = node.deepCopy();
        for (String path : paths) {
            maskPath(masked, Arrays.asList(path.split("\\.")), maskValue);
        }
        return removeNullsRecursively(masked);
    }

    private void maskPath(JsonNode node, List<String> path, String maskValue) {
        if (node == null || path.isEmpty()) return;

        String current = path.get(0);
        List<String> rest = path.subList(1, path.size());

        if (node instanceof ObjectNode) {
            JsonNode child = node.get(current);
            if (child != null) {
                if (rest.isEmpty()) {
                    ((ObjectNode) node).put(current, maskValue);
                } else {
                    maskPath(child, rest, maskValue);
                }
            }
        } else if (node instanceof ArrayNode) {
            for (JsonNode item : node) {
                maskPath(item, path, maskValue);
            }
        }
    }

    private JsonNode removeNullsRecursively(JsonNode node) {
        if (node == null || node.isNull()) return MissingNode.getInstance();

        if (node instanceof ObjectNode) {
            ObjectNode obj = (ObjectNode) node;
            List<String> toRemove = new ArrayList<>();
            obj.fieldNames().forEachRemaining(field -> {
                JsonNode cleaned = removeNullsRecursively(obj.get(field));
                if (cleaned.isMissingNode() || cleaned.isNull()) {
                    toRemove.add(field);
                } else {
                    obj.set(field, cleaned);
                }
            });
            toRemove.forEach(obj::remove);
            return obj;
        }

        if (node instanceof ArrayNode) {
            ArrayNode arr = (ArrayNode) node;
            ArrayNode cleanedArr = new ArrayNode(JsonNodeFactory.instance);
            for (JsonNode item : arr) {
                JsonNode cleaned = removeNullsRecursively(item);
                if (!cleaned.isMissingNode() && !cleaned.isNull()) {
                    cleanedArr.add(cleaned);
                }
            }
            return cleanedArr;
        }

        return node;
    }
}

// src/main/java/com/example/demo/config/WebFluxHandlerConfig.java
package com.example.demo.config;

import com.example.demo.handler.AnnotationDrivenMaskingHandler;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.codec.ServerCodecConfigurer;
import org.springframework.web.reactive.config.WebFluxConfigurer;
import org.springframework.web.reactive.result.method.annotation.RequestMappingHandlerAdapter;

import javax.annotation.PostConstruct;

@Configuration
public class WebFluxHandlerConfig implements WebFluxConfigurer {

    private final RequestMappingHandlerAdapter adapter;
    private final ServerCodecConfigurer codecConfigurer;
    private final ObjectMapper objectMapper;

    public WebFluxHandlerConfig(RequestMappingHandlerAdapter adapter, ServerCodecConfigurer codecConfigurer, ObjectMapper objectMapper) {
        this.adapter = adapter;
        this.codecConfigurer = codecConfigurer;
        this.objectMapper = objectMapper;
    }

    @PostConstruct
    public void init() {
        adapter.setCustomHandlerResultHandlers(
                java.util.List.of(new AnnotationDrivenMaskingHandler(codecConfigurer.getWriters(), objectMapper))
        );
    }
}

package com.example.demo;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.reactive.AutoConfigureWebTestClient;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.web.reactive.server.WebTestClient;

@SpringBootTest
@AutoConfigureWebTestClient
public class MaskingHandlerTest {

    @Autowired
    private WebTestClient webTestClient;

    @Test
    void testMaskingAndNullRemoval() {
        webTestClient.get()
                .uri("/api/user")
                .exchange()
                .expectStatus().isOk()
                .expectBody()
                .jsonPath("$.user.password").isEqualTo("***")
                .jsonPath("$.profile.ssn").isEqualTo("***")
                .jsonPath("$.user.email").doesNotExist()
                .jsonPath("$.profile.dob").doesNotExist()
                .jsonPath("$.metadata").doesNotExist();
    }
}
